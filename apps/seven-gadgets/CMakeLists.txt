cmake_minimum_required(VERSION 3.16)
project(seven-gadgets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network UiPlugin)
foreach(fw Config FrameworkIntegration WindowSystem WidgetsAddons CoreAddons Completion KIO)
    find_package(KF6${fw} REQUIRED)
endforeach()
find_package(AeroQt REQUIRED)
add_executable(seven-gadgets main.cpp)
target_include_directories(seven-gadgets PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../seven-common)
target_link_libraries(seven-gadgets PRIVATE Qt6::Widgets Qt6::Network KF6::WindowSystem AeroQt::AeroQt)
install(TARGETS seven-gadgets RUNTIME DESTINATION bin)
# the genuine Windows 7 gadget pictures (from the Windows 7 ISO, "Program Files/Windows Sidebar")
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/data)
    install(DIRECTORY data/ DESTINATION share/seven-gadgets)
endif()
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/seven-weather/images)
    install(DIRECTORY seven-weather/images DESTINATION share/seven-gadgets/weather)
endif()
